## 2. Architecture: The Linux Driver Model

In modern Linux, hardware is never hardcoded. The kernel's **Platform Bus** dynamically marries hardware nodes declared in the Device Tree with registered drivers:

```
+------------------------------------+      +------------------------------------+
|         Device Tree (.dts)         |      |       Platform Driver (.c)         |
|                                    |      |                                    |
|  foo_sensor: sensor@1000 {         |      |  static struct of_device_id match[]|
|    compatible = "acme,foo-sensor"; | <==> |    { .compatible = "acme,foo..." } |
|    reg = <0x1000 0x100>;           |Match |  };                                |
|    interrupts = <0 42 4>;          |      |  module_platform_driver(foo_drv);  |
|  };                                |      +-----------------+------------------+
+------------------------------------+                        |
                                                              v
                                              Matches! Kernel calls: foo_probe()
                                                              |
                                                              v
                                              Registers /dev/foo char device
                                                              |
                                                              v
                                       Userspace HAL calls: open() / ioctl()
```

## 4. Writing the Driver: `foo_driver.c`

Let's write a production-grade character device driver that binds to a Device Tree node and provides `read`, `write`, and `unlocked_ioctl`:

<CodeAnnotator
  title="foo_driver.c: Device-Tree Bound Platform Character Driver"
  code={`#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/of.h>
#include <linux/fs.h>
#include <linux/uaccess.h>
#include <linux/miscdevice.h>

#define FOO_IOCTL_MAGIC 'F'
#define FOO_RESET _IO(FOO_IOCTL_MAGIC, 0)
#define FOO_GET_STATUS _IOR(FOO_IOCTL_MAGIC, 1, int)

struct foo_dev {
    struct miscdevice misc;
    void __iomem *regs;
    int status;
    struct mutex lock;
};

static ssize_t foo_read(struct file *f, char __user *buf, size_t count, loff_t *ppos) {
    struct foo_dev *dev = container_of(f->private_data, struct foo_dev, misc);
    int val = dev->status;
    if (copy_to_user(buf, &val, sizeof(val)))
        return -EFAULT;
    return sizeof(val);
}

static long foo_ioctl(struct file *f, unsigned int cmd, unsigned long arg) {
    struct foo_dev *dev = container_of(f->private_data, struct foo_dev, misc);
    switch (cmd) {
        case FOO_RESET:
            mutex_lock(&dev->lock);
            dev->status = 0;
            mutex_unlock(&dev->lock);
            return 0;
        case FOO_GET_STATUS:
            if (copy_to_user((int __user *)arg, &dev->status, sizeof(int)))
                return -EFAULT;
            return 0;
        default:
            return -ENOTTY; // Standard POSIX error for unknown ioctls
    }
}

static const struct file_operations foo_fops = {
    .owner          = THIS_MODULE,
    .read           = foo_read,
    .unlocked_ioctl = foo_ioctl,
};

static int foo_probe(struct platform_device *pdev) {
    struct foo_dev *dev;
    int ret;

    dev = devm_kzalloc(&pdev->dev, sizeof(*dev), GFP_KERNEL);
    if (!dev) return -ENOMEM;

    mutex_init(&dev->lock);
    dev->status = 42;

    dev->misc.minor = MISC_DYNAMIC_MINOR;
    dev->misc.name = "foo_device";
    dev->misc.fops = &foo_fops;

    ret = misc_register(&dev->misc);
    if (ret) return ret;

    platform_set_drvdata(pdev, dev);
    pr_info("foo_driver: successfully probed and registered /dev/foo_device\\n");
    return 0;
}

static int foo_remove(struct platform_device *pdev) {
    struct foo_dev *dev = platform_get_drvdata(pdev);
    misc_deregister(&dev->misc);
    pr_info("foo_driver: removed\\n");
    return 0;
}

static const struct of_device_id foo_of_match[] = {
    { .compatible = "acme,foo-sensor" },
    { /* sentinel */ }
};
MODULE_DEVICE_TABLE(of, foo_of_match);

static struct platform_driver foo_driver = {
    .probe = foo_probe,
    .remove = foo_remove,
    .driver = {
        .name = "foo_sensor",
        .of_match_table = foo_of_match,
    },
};
module_platform_driver(foo_driver);

MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("Platform Character Driver with Device Tree Matching");`}

## 5. The Device Tree Binding

Add the corresponding device tree node to your device DTS file (`arch/arm64/boot/dts/...`):

```dts
// Hardware description node
foo_node: sensor@10000000 {
    compatible = "acme,foo-sensor";
    reg = <0x0 0x10000000 0x0 0x1000>;
    status = "okay";
};
```

Compile the DTS into a `.dtb` using the Device Tree Compiler:

```bash
dtc -I dts -O dtb -o my_device.dtb my_device.dts
```

## 6. Testing Driver Logic with KUnit

Before touching real hardware, verify driver calculations and edge cases using in-kernel KUnit tests:

```c
// foo_driver_test.c
#include <kunit/test.h>

static void test_foo_ioctl_macros(struct kunit *test) {
    // Assert ioctl command numbers encode proper size and magic
    KUNIT_EXPECT_EQ(test, _IOC_TYPE(FOO_GET_STATUS), 'F');
    KUNIT_EXPECT_EQ(test, _IOC_SIZE(FOO_GET_STATUS), sizeof(int));
}

static struct kunit_case foo_driver_test_cases[] = {
    KUNIT_CASE(test_foo_ioctl_macros),
    {}
};

static struct kunit_suite foo_driver_suite = {
    .name = "foo_driver_suite",
    .test_cases = foo_driver_test_cases,
};
kunit_test_suite(foo_driver_suite);
```
